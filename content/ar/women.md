المرأة في الإسلام طرفٌ في العقد لا استثناءٌ منه؛ كرامتها ليست منّةً تُمنح ولا رقماً يُوزن، بل حقٌّ ثابت واجب. ولذلك لا تُقاس وضعيتُها بمقولةٍ عابرة، بل بنصوصٍ ثبتت وسوّت بين الطرفين في الأصل، ورتّبت الحقوق بالتزامات لا بالنِّسب وحدها.

:::Ayah text="وَالْمُؤْمِنُونَ وَالْمُؤْمِنَاتُ بَعْضُهُمْ أَوْلِيَاءُ بَعْضٍ" ref="التوبة — ٧١" tr="مؤمنٌ ومؤمنة: طرفان في عهدٍ واحد، لا درجتان في كرامة."
:::

:::Section title="ستة حقوق ثابتة"
:::Cards cols="3"
:::Card icon="📚" title="الحق في التعليم"
طلب العلم فريضة، والمرأة فيه كغيرها؛ «وَقُل رَّبِّ زِدْنِي عِلْمًا» (طه — ١١٤)، ومن سلك طريقاً يلتمس فيه علماً سهّل الله له طريقاً إلى الجنة (رواه مسلم).
:::
:::Card icon="💰" title="الحق في المال والميراث"
تملك وتبيع وتتصّدق وتشترط، ومالها لا يُغصب عليها؛ ولها نصيبٌ في الميراث بينةً في كتاب الله، ولها النفقة والسكنى.
:::
:::Card icon="💍" title="اختيار الزوج"
«فَلَا تَعْضُلُوهُنَّ أَن يَنكِحْنَ أَزْوَاجَهُنَّ إِذَا تَرَاضَوْا بَيْنَهُم بِالْمَعْرُوفِ» (البقرة — ٢٣٢)؛ فالرضاء شرطٌ، والمنع بغير سببٍ محظور.
:::
:::Card icon="🏡" title="الأمومة والأسرة"
وأوصينا الإنسان بوالديه: «حَمَلَتْهُ أُمُّهُ وَهْنًا عَلَىٰ وَهْنٍ» (لقمان — ١٤)؛ فالرعايةُ والحنانُ ثابتان، والقوامةُ فيها حنانٌ لا سلطنة.
:::
:::Card icon="🛡️" title="الحماية والعدالة"
النفقة والسكنى والهيبة على الزوج، والعدلُ مأمور به: «وَعَاشِرُوهُنَّ بِالْمَعْرُوفِ» (النساء — ١٩)، وقد قال ﷺ: «خَيْرُكُمْ خَيْرُكُمْ لِأَهْلِهِ» (رواه الترمذي).
:::
:::Card icon="🤝" title="المشاركة في الخير"
خديجة رضي الله عنها نصرةٌ ومالٌ ومشورة، وأسماءٌ في الهجرة، ونساءٌ في البيعة؛ والزكاة والصدقة حقٌّ لهنّ وللفقراء.
:::
:::
:::

:::Section title="حقوقٌ لا مُكرَمات"
المسألة ليست في عدد الحقوق المكتوبة، بل في نظامٍ متوازن: إنفاقٌ على جهة، وميراثٌ على جهة، وتعلّمٌ واختيارٌ وحمايةٌ ومشاركة. ومن قاس النصيبين من غير أن ينظر في الالتزامات أخذ نصف الحقيقة وترك النصف؛ والقاعدة أنّ ما ثبت بالنصّ لا يُلغى بالرأي، وما تغيّر في العرف يُراجَع في الفروع لا في الأصول.

:::Ayah text="لِلرِّجَالِ نَصِيبٌ مِّمَّا تَرَكَ الْوَالِدَانِ وَالْأَقْرَبُونَ وَلِلنِّسَاءِ نَصِيبٌ مِّمَّا تَرَكَ الْوَالِدَانِ وَالْأَقْرَبُونَ مِمَّا قَلَّ مِنْهُ أَوْ كَثُرَ" ref="النساء — ٧" tr="للمرأة نصيبٌ ثابت كما للرجل، وعلّةُ الإرث الرَّحم والعطاء لا الذكورة."
:::
:::

:::Section title="أربعة أسئلة شائعة"
:::FAQ
:::QA q="هل الميراث يُقصّر على المرأة؟"
النصّ قطعي: «يُوصِيكُمُ اللَّهُ فِي أَوْلَادِكُمْ لِلذَّكَرِ مِثْلُ حَظِّ الْأُنثَيَيْنِ» (النساء — ١١). والمقارنة الصحيحة تنظر إلى مجموع الالتزامات: النفقة والسكنى على الرجل، والمهر ملكٌ للمرأة، ولا نفقةً عليها في مقابل ذلك. ومن أخذ النصيبين من غير هذا السياق أخذ نصف الحقيقة وترك النصف.
:::
:::QA q="هل يجوز للمرأة أن تتعلّم؟"
العلم فريضة على المسلم والمسلمة، ومنها قول ﷺ: «مَنْ سَلَكْ طَرِيقًا يَلْتَمِسُ فِيهِ عِلْمًا سَهَّلَ اللَّهُ لَهُ بِهِ طَرِيقًا إِلَى الْجَنَّةِ» (رواه مسلم). وطلب العلم شرطٌ لصلاح الفرد والأسرة والمجتمع، ولم يُستثنِ أحدٌ بحجّة الجنس.
:::
:::QA q="هل الحجاب اختياري أم فرض؟"
الحكم الشرعي فرضٌ على من تكليف: «وَلْيَضْرِبْنَ بِخُمُرِهِمْ عَلَىٰ جُيُوبِهِنَّ» (النور — ٣١). وأمّا التطبيق فبالحكمة لا بالقوّة، فالدين ينفي الإكراه: «لَا إِكْرَاهَ فِي الدِّينِ» (البقرة — ٢٥٦)؛ فتُدعى بالقول الحسن والمثال لا بالضرب.
:::
:::QA q="هل يجوز عمل المرأة؟"
يجوز بشرطين: ألا يُخالف الطاعة، وألا يُضرّ بأهلها؛ ولها أن تكسب وتملك وتُنفق. وقد جاءت السيرة بعمل المرأة الموقّر: خديجة رضي الله عنها تاجرٌ كريم يُفوّضها زبناؤها، فما منعها الإسلام من الرزق بل أكرمه وشرّع لها أن تُحسن إليه.
:::
